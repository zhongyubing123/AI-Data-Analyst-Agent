库：ecommerce

表 dws_item_hot(item_id, item_name, category, brand, pv_count, fav_count, cart_count, buy_count, buyer_count, sales_amount)
  商品维度。金额只统计购买行为。

表 dws_brand_sales(brand, buy_count, buyer_count, sales_amount, avg_amount)
  品牌维度。avg_amount 是客单价。

表 dws_category_sales(category, buy_count, item_count, buyer_count, sales_amount, avg_amount)
  品类维度。item_count 是该品类商品数。

表 dws_price_band(price_level, item_count, pv_count, buy_count, buyer_count, sales_amount, convert_rate)
  价格带维度。convert_rate 是购买次数除以浏览次数。

表 dws_daily_trend(dt, pv_count, cart_count, buy_count, buyer_count, sales_amount)
  每日趋势。dt 形如 2025-08-01。

表 dws_user_value(user_id, pv_count, fav_count, cart_count, buy_count, total_amount, first_dt, last_dt, value_level)
  用户维度。value_level 取值为 高价值、中价值、一次购买、仅浏览。
